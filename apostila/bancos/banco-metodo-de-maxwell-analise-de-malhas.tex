\subsubsection*{Banco complementar --- Método de Maxwell (análise de malhas)}

\begin{atencao}[Organização do banco]
A montagem por inspeção segue duas regras fixas: a \textbf{diagonal}
$R_{kk}$ é a soma de todas as resistências da malha $k$; o termo \textbf{fora
da diagonal} $R_{jk}$ é o \emph{negativo} da resistência compartilhada entre as
malhas $j$ e $k$ (zero, se não houver). No vetor independente entram as fontes
de tensão, positivas quando impulsionam a corrente no sentido adotado. O N3
trata de fontes na periferia, fontes dependentes e forma quadrática; o N4 exige
demonstração, diagnóstico de singularidade, algoritmo e dualidade.
\end{atencao}

\blocofixacao

\begin{questao}[1]{}
Uma malha contém resistores de $\SI{4}{\ohm}$, $\SI{6}{\ohm}$ e
$\SI{10}{\ohm}$, sendo o de $\SI{10}{\ohm}$ compartilhado com a malha vizinha.
Determine $R_{11}$ e $R_{12}$.
\resposta{$R_{11}=\SI{20}{\ohm}$; $R_{12}=\SI{-10}{\ohm}$}
\resolucao{\textbf{Diagonal:} soma de \emph{todas} as resistências da malha,
inclusive a compartilhada:
\[
R_{11}=4+6+10=\SI{20}{\ohm}.
\]
\textbf{Fora da diagonal:} o negativo da resistência comum:
\[
R_{12}=-\SI{10}{\ohm}.
\]
O sinal negativo aparece porque, com ambas as correntes de malha adotadas no
mesmo sentido (horário, por convenção), elas percorrem o resistor comum em
sentidos \emph{opostos}.}
\end{questao}

\begin{questao}[1]{}
Um circuito planar tem $b=\SI{9}{}$ ramos e $n=\SI{5}{}$ nós. Determine o
número de malhas independentes.
\resposta{$m=5$}
\resolucao{
\[
m=b-n+1=9-5+1=5.
\]
São, portanto, $5$ equações de malha --- contra $n-1=4$ equações nodais. Para
\emph{este} circuito a análise nodal produz o sistema menor.\\
A contagem deve ser feita \emph{antes} de escolher o método: ela custa segundos
e pode poupar um sistema inteiro.}
\end{questao}

\begin{questao}[1]{}
Por que se adota o mesmo sentido (usualmente horário) para todas as correntes
de malha?
\begin{alternativas}[2]
\item porque é o sentido físico real das correntes
\item porque torna todos os termos fora da diagonal negativos
\item porque é exigido pela lei de Kirchhoff
\item porque evita resultados negativos
\end{alternativas}
\resposta{(b)}
\resolucao{A escolha é \textbf{arbitrária} --- qualquer conjunto de sentidos
leva ao mesmo resultado físico. Adotar todos horários é uma \emph{convenção de
conveniência}: garante que toda corrente compartilhada seja percorrida em
sentidos opostos pelas duas malhas, tornando \emph{todos} os $R_{jk}$
negativos. A matriz fica simétrica com padrão de sinais previsível, e a
montagem por inspeção vira mecânica.\\
As alternativas (a) e (d) são falsas: correntes de malha são variáveis
auxiliares, não correntes físicas, e podem perfeitamente resultar negativas.}
\end{questao}

\begin{questao}[1]{}
Uma malha contém uma fonte de $\SI{24}{\volt}$ que impulsiona a corrente no
sentido adotado, e outra de $\SI{9}{\volt}$ que se opõe. Determine o termo
independente dessa malha.
\resposta{$\SI{15}{\volt}$}
\resolucao{
\[
b_k=+24-9=\SI{15}{\volt}.
\]
A regra de sinal: some as fontes que \emph{empurram} a corrente no sentido
adotado e subtraia as que se opõem. Fontes que pertencem apenas àquela malha
entram uma única vez; se a fonte estiver em um ramo compartilhado, ela aparece
--- com sinais adequados --- nas equações das \emph{duas} malhas.}
\end{questao}

\begin{questao}[1]{}
Resolvido um sistema, obteve-se $I_1=\SI{2}{\ampere}$ e
$I_2=\SI{1}{\ampere}$. Determine a corrente real em um resistor pertencente
apenas à malha 1 e em um resistor compartilhado.
\resposta{$\SI{2}{\ampere}$ e $\SI{1}{\ampere}$}
\resolucao{\textbf{Resistor exclusivo da malha 1:} é percorrido apenas por
$I_1$, logo conduz $\SI{2}{\ampere}$.\\
\textbf{Resistor compartilhado:} é percorrido pelas duas correntes em sentidos
opostos, logo conduz a \emph{diferença}:
\[
I=I_1-I_2=2-1=\SI{1}{\ampere},
\]
no sentido de $I_1$.\\
Este é o passo de \emph{expansão} do método: as correntes de malha são
variáveis intermediárias, e as correntes reais dos ramos precisam ser
recuperadas ao final.}
\end{questao}

\begin{questao}[1]{}
Resolva o sistema
$\begin{bmatrix}30&-20\\-20&40\end{bmatrix}
\begin{bmatrix}I_1\\I_2\end{bmatrix}=
\begin{bmatrix}40\\0\end{bmatrix}$.
\resposta{$I_1=\SI{2}{\ampere}$; $I_2=\SI{1}{\ampere}$}
\resolucao{Da segunda linha: $-20I_1+40I_2=0\Rightarrow I_1=2I_2$.
Substituindo na primeira:
\[
30(2I_2)-20I_2=40 \;\Longrightarrow\; 40I_2=40 \;\Longrightarrow\;
I_2=\SI{1}{\ampere},\quad I_1=\SI{2}{\ampere}.
\]
\textbf{Conferência estrutural:} a matriz é simétrica, com diagonal positiva e
termos fora da diagonal negativos --- como deve ser em rede passiva. E é
diagonal dominante ($30>20$; $40>20$), o que garante solução única.}
\end{questao}

\begin{questao}[1]{}
Na matriz de malhas de uma rede puramente resistiva, o elemento $R_{kk}$
representa:
\begin{alternativas}[2]
\item apenas as resistências exclusivas da malha $k$
\item a soma de todas as resistências da malha $k$
\item a resistência compartilhada com a malha vizinha
\item a resistência equivalente vista pela fonte
\end{alternativas}
\resposta{(b)}
\resolucao{A diagonal reúne \textbf{todas} as resistências percorridas por
$I_k$ --- exclusivas e compartilhadas. As compartilhadas aparecem duas vezes na
matriz: na diagonal (positivas, em ambas as malhas que as contêm) e fora dela
(negativas).\\
Esquecer a resistência comum na diagonal é o erro de montagem mais frequente do
método, e produz sistemas que não satisfazem a dominância diagonal --- sinal de
alerta útil.}
\end{questao}

\blocoaplicacao

\begin{questao}[2]{}
Uma malha 1 contém uma fonte de $\SI{40}{\volt}$ e $R_a=\SI{10}{\ohm}$; ela
compartilha $R_c=\SI{20}{\ohm}$ com a malha 2, que contém ainda
$R_b=\SI{20}{\ohm}$. Monte o sistema e resolva.
\resposta{$I_1=\SI{2}{\ampere}$; $I_2=\SI{1}{\ampere}$}
\resolucao{\textbf{Por inspeção:}
\[
R_{11}=10+20=30;\quad R_{22}=20+20=40;\quad R_{12}=R_{21}=-20.
\]
\[
\begin{bmatrix}30&-20\\-20&40\end{bmatrix}
\begin{bmatrix}I_1\\I_2\end{bmatrix}
=\begin{bmatrix}40\\0\end{bmatrix}
\;\Longrightarrow\;
I_1=\SI{2}{\ampere},\ I_2=\SI{1}{\ampere}.
\]
\textbf{Verificação pela LKT na malha 1:}
$10(2)+20(2-1)=20+20=\SI{40}{\volt}$ \checkmark. A malha 2, sem fonte, deve
fechar em zero: $20(1)+20(1-2)=0$ \checkmark.}
\end{questao}

\begin{questao}[2]{}
Duas malhas compartilham $R_c=\SI{6}{\ohm}$. A malha 1 tem
$R_a=\SI{6}{\ohm}$ e fonte de $\SI{36}{\volt}$; a malha 2 tem
$R_b=\SI{6}{\ohm}$ e fonte de $\SI{18}{\volt}$, ambas impulsionando as
correntes no sentido adotado. Determine $I_1$, $I_2$ e a corrente em $R_c$.
\resposta{$I_1=\SI{5}{\ampere}$; $I_2=\SI{4}{\ampere}$; $I_{R_c}=\SI{1}{\ampere}$}
\resolucao{
\[
\begin{bmatrix}12&-6\\-6&12\end{bmatrix}
\begin{bmatrix}I_1\\I_2\end{bmatrix}
=\begin{bmatrix}36\\18\end{bmatrix}.
\]
Determinante: $144-36=108$. Por Cramer:
\[
I_1=\frac{36(12)+6(18)}{108}=\frac{540}{108}=\SI{5}{\ampere};
\qquad
I_2=\frac{12(18)+6(36)}{108}=\frac{432}{108}=\SI{4}{\ampere}.
\]
\[
I_{R_c}=I_1-I_2=\SI{1}{\ampere}.
\]
\textbf{Observação:} apesar de a fonte da malha 1 ser o dobro da outra, as
correntes de malha ficaram próximas ($5$ e $4$) --- o acoplamento pelo resistor
comum arrasta uma malha na direção da outra.}
\end{questao}

\begin{questao}[2]{}
No circuito anterior, determine a potência de cada fonte e de cada resistor, e
verifique o balanço.
\resposta{Fontes: $\SI{180}{\watt}$ e $\SI{72}{\watt}$; resistores:
$150$, $96$ e $\SI{6}{\watt}$}
\resolucao{\textbf{Fontes:} cada uma entrega $V_kI_k$ (a corrente de malha é a
que a atravessa, pois as fontes não estão em ramos compartilhados):
\[
P_1=36(5)=\SI{180}{\watt};\qquad P_2=18(4)=\SI{72}{\watt};
\qquad \text{total }\SI{252}{\watt}.
\]
\textbf{Resistores:}
\[
P_{R_a}=6(5)^{2}=150;\quad
P_{R_b}=6(4)^{2}=96;\quad
P_{R_c}=6(1)^{2}=\SI{6}{\watt};
\qquad \text{total }\SI{252}{\watt}.\ \checkmark
\]
\textbf{Atenção ao resistor compartilhado:} sua potência usa a corrente
\emph{real} $(I_1-I_2)$, não $I_1$ nem $I_2$. Usar uma corrente de malha aqui
daria $\SI{150}{\watt}$ em vez de $\SI{6}{\watt}$ --- e o balanço não
fecharia, denunciando o erro.}
\end{questao}

\begin{questao}[2]{}
Uma malha 1 com fonte de $\SI{30}{\volt}$ e $R_a=\SI{5}{\ohm}$ compartilha
$R_c=\SI{10}{\ohm}$ com a malha 2, que tem $R_b=\SI{5}{\ohm}$ e uma fonte de
$\SI{20}{\volt}$ \textbf{opondo-se} ao sentido adotado. Monte o sistema.
\resposta{$\begin{bmatrix}15&-10\\-10&15\end{bmatrix}
\begin{bmatrix}I_1\\I_2\end{bmatrix}=
\begin{bmatrix}30\\-20\end{bmatrix}$}
\resolucao{\textbf{Diagonais:} $R_{11}=5+10=15$; $R_{22}=5+10=15$.\\
\textbf{Fora da diagonal:} $R_{12}=R_{21}=-10$.\\
\textbf{Termos independentes:} a fonte da malha 1 impulsiona
($+30$); a da malha 2 se opõe ($-20$).\\
Resolvendo: $\det=225-100=125$;
\[
I_1=\frac{30(15)+10(-20)}{125}=\frac{250}{125}=\SI{2}{\ampere};
\qquad
I_2=\frac{15(-20)+10(30)}{125}=\frac{0}{125}=\SI{0}{\ampere}.
\]
\textbf{Resultado notável:} $I_2=0$. A fonte de $\SI{20}{\volt}$ equilibra
exatamente a tensão que a malha 1 induz sobre $R_c$
($10\times2=\SI{20}{\volt}$), e a malha 2 fica sem corrente.}
\end{questao}

\begin{questao}[2]{}
Três malhas dispostas \textbf{em linha}: a malha 1 compartilha
$\SI{10}{\ohm}$ com a 2, e a 2 compartilha $\SI{10}{\ohm}$ com a 3; as malhas
1 e 3 não se tocam. As resistências próprias adicionais são $\SI{10}{\ohm}$
em cada malha, e há uma fonte de $\SI{60}{\volt}$ na malha 1. Monte a matriz.
\resposta{$\begin{bmatrix}20&-10&0\\-10&30&-10\\0&-10&20\end{bmatrix}$,
com $\mathbf{b}=(60;0;0)$}
\resolucao{\textbf{Diagonais:} malha 1: $10+10=20$; malha 2:
$10+10+10=30$ (ela tem \emph{duas} resistências compartilhadas); malha 3:
$10+10=20$.\\
\textbf{Fora da diagonal:} $R_{12}=R_{21}=-10$ e $R_{23}=R_{32}=-10$. Como as
malhas 1 e 3 não compartilham nada, $R_{13}=R_{31}=\mathbf{0}$.\\
A matriz é \textbf{tridiagonal} --- padrão típico de topologias em cadeia, e a
mesma estrutura que aparece na análise nodal de redes em escada.\\
Resolvendo: $I_1=\SI{3.75}{\ampere}$, $I_2=\SI{1.5}{\ampere}$,
$I_3=\SI{0.75}{\ampere}$. Note a atenuação progressiva ao longo da cadeia.}
\end{questao}

\begin{questao}[2]{}
Verifique o resultado da questão anterior aplicando a LKT à malha 3.
\resposta{Fecha em zero}
\resolucao{A malha 3 não tem fonte, então a soma das quedas deve ser nula:
\[
10\,I_3+10\,(I_3-I_2)=10(0{,}75)+10(0{,}75-1{,}5)=7{,}5-7{,}5=0.\ \checkmark
\]
\textbf{Estratégia de verificação:} confira sempre a malha \emph{sem fonte} ---
ela oferece o teste mais limpo, pois qualquer erro aparece como um resíduo não
nulo, sem precisar comparar com um valor de referência.\\
Como teste adicional, a malha 2 dá
$10(1{,}5)+10(1{,}5-3{,}75)+10(1{,}5-0{,}75)=15-22{,}5+7{,}5=0$ \checkmark.}
\end{questao}

\begin{questao}[2]{}
Explique por que a matriz de malhas de uma rede puramente resistiva é
simétrica, e o que isso permite economizar na montagem.
\resposta{$R_{jk}=R_{kj}=-R_{comum}$; basta montar metade da matriz}
\resolucao{\textbf{Razão:} o termo $R_{jk}$ é determinado pela resistência
\emph{compartilhada} entre as malhas $j$ e $k$ --- e "compartilhada" é uma
relação simétrica. O mesmo resistor produz o mesmo valor nas duas posições.\\
\textbf{Economia:} basta montar a diagonal e o triângulo superior; o inferior
sai por espelhamento. Para $m$ malhas, isso reduz de $m^{2}$ para
$m(m+1)/2$ termos --- de $16$ para $10$ com quatro malhas.\\
\textbf{Uso como teste:} se a matriz montada sair assimétrica em rede puramente
resistiva, há erro de montagem. A única situação em que a assimetria é
\emph{legítima} é a presença de fontes dependentes --- caso tratado adiante.}
\end{questao}

\begin{questao}[2]{}
Uma fonte de tensão está em um ramo \textbf{compartilhado} por duas malhas.
Explique como ela entra no sistema.
\resposta{Aparece nos termos independentes das \emph{duas} malhas, com sinais
opostos}
\resolucao{O ramo comum é percorrido por $I_1$ em um sentido e por $I_2$ no
sentido contrário. Ao escrever a LKT de cada malha, a mesma fonte é atravessada
em sentidos opostos --- e entra com sinais opostos nos dois termos
independentes.\\
\textbf{Exemplo:} fonte de $\SI{12}{\volt}$ no ramo comum, impulsionando
$I_1$:
\[
b_1=+12
\qquad\text{e}\qquad
b_2=-12.
\]
\textbf{Contraste importante:} isso vale para fonte de \emph{tensão}. Uma fonte
de \emph{corrente} no ramo comum não pode ser tratada assim --- sua tensão é
desconhecida --- e exige a técnica da supermalha.}
\end{questao}

\blocoaprofundamento

\begin{questao}[3]{}
Uma fonte de corrente de $\SI{2}{\ampere}$ está em um ramo que pertence
\textbf{apenas} à malha 2 (periferia do circuito). A malha 1 tem fonte de
$\SI{40}{\volt}$ e $R_a=\SI{10}{\ohm}$, e compartilha $R_c=\SI{10}{\ohm}$ com
a malha 2.
\begin{itensq}
\item Explique por que \emph{não} é necessário formar supermalha.
\item Determine $I_1$ e a corrente em $R_c$.
\item Compare o esforço com o caso em que a fonte estivesse no ramo comum.
\end{itensq}
\resposta{(b) $I_1=\SI{3}{\ampere}$; $I_{R_c}=\SI{1}{\ampere}$}
\resolucao{(a) Se a fonte de corrente pertence a \emph{uma única} malha, ela
\textbf{impõe diretamente} o valor daquela corrente de malha:
$I_2=\SI{2}{\ampere}$. A incógnita desaparece, e a equação da malha 2 não
precisa ser escrita --- ela serviria apenas para descobrir a tensão sobre a
fonte, se fosse pedida.\\
(b) Resta uma única equação, a da malha 1:
\[
40=10I_1+10(I_1-I_2)=20I_1-10(2)
\;\Longrightarrow\;
20I_1=60 \;\Longrightarrow\; I_1=\SI{3}{\ampere}.
\]
\[
I_{R_c}=I_1-I_2=3-2=\SI{1}{\ampere}.
\]
(c) \textbf{Comparação.} Fonte na periferia: o sistema \emph{encolhe} de
$2\times2$ para $1\times1$ --- a fonte de corrente \emph{ajuda}.\\
Fonte no ramo comum: nenhuma corrente de malha fica determinada, e é preciso
formar a supermalha (uma LKT contornando as duas) mais a equação de restrição
$I_1-I_2=I_{fonte}$. O sistema continua $2\times2$, com montagem mais delicada.\\
\textbf{Regra prática:} ao escolher os contornos das malhas, procure fazer cada
fonte de corrente pertencer a uma malha só. Muitas vezes isso é possível e
simplifica bastante o problema.}
\end{questao}

\begin{questao}[3]{}
Um circuito de duas malhas tem $R_a=R_b=\SI{5}{\ohm}$ e
$R_c=\SI{5}{\ohm}$ compartilhado, com fonte de $\SI{20}{\volt}$ na malha 1. Na
malha 2 há uma fonte \textbf{dependente} de tensão, de valor
$2{,}5\,I_1$ volts, impulsionando $I_2$.
\begin{itensq}
\item Monte o sistema e resolva.
\item Comente a estrutura da matriz.
\end{itensq}
\resposta{$I_1=\SI{3.2}{\ampere}$; $I_2=\SI{2.4}{\ampere}$; matriz
\textbf{assimétrica}}
\resolucao{(a) \textbf{Malha 1:} $10I_1-5I_2=20$.\\
\textbf{Malha 2:} $-5I_1+10I_2=2{,}5\,I_1$. A fonte dependente é função de uma
incógnita, então seu termo \emph{migra para a matriz}:
\[
-7{,}5\,I_1+10I_2=0.
\]
\[
\begin{bmatrix}10&-5\\-7{,}5&10\end{bmatrix}
\begin{bmatrix}I_1\\I_2\end{bmatrix}
=\begin{bmatrix}20\\0\end{bmatrix}.
\]
Da segunda linha, $I_2=0{,}75\,I_1$. Na primeira:
$10I_1-3{,}75I_1=20\Rightarrow I_1=\SI{3.2}{\ampere}$ e
$I_2=\SI{2.4}{\ampere}$. Corrente em $R_c$: $\SI{0.8}{\ampere}$.\\
(b) \textbf{A matriz perdeu a simetria} ($R_{12}=-5$ mas
$R_{21}=-7{,}5$). Isso não é erro: fontes dependentes quebram a
\textbf{reciprocidade}, exatamente como ocorre na matriz de condutâncias da
análise nodal.\\
\textbf{Consequências:} \emph{(i)} não vale mais montar só metade da matriz;
\emph{(ii)} perde-se a garantia automática de dominância diagonal, e a matriz
pode até tornar-se singular para certos valores do coeficiente; \emph{(iii)}
elemento ativo e reciprocidade são mutuamente exclusivos --- é a assinatura
matemática de um amplificador.}
\end{questao}

\begin{questao}[3]{}
No circuito de duas malhas com $R_a=R_b=R_c=\SI{6}{\ohm}$ e fontes de
$\SI{36}{\volt}$ e $\SI{18}{\volt}$ ($I_1=\SI{5}{\ampere}$,
$I_2=\SI{4}{\ampere}$), calcule a potência total pela forma quadrática
$\mathbf{I}^{\!\top}\mathbf{R}\,\mathbf{I}$ e confronte com a soma direta.
\resposta{$\SI{252}{\watt}$ pelos dois caminhos}
\resolucao{\textbf{Forma quadrática:}
\[
\mathbf{I}^{\!\top}\mathbf{R}\,\mathbf{I}
=R_{11}I_1^{2}+R_{22}I_2^{2}+2R_{12}I_1I_2
=12(25)+12(16)+2(-6)(5)(4),
\]
\[
=300+192-240=\SI{252}{\watt}.
\]
\textbf{Soma direta:} $6(5)^{2}+6(4)^{2}+6(1)^{2}=150+96+6=\SI{252}{\watt}$
\checkmark.\\
\textbf{Interpretação do termo cruzado:} os $\SI{-240}{\watt}$ não são potência
negativa --- eles \emph{corrigem} a dupla contagem de $R_c$, que aparece nas
duas diagonais. De fato, $R_c$ é contabilizado como
$6(25)+6(16)-240=246-240=\SI{6}{\watt}$, exatamente
$6(I_1-I_2)^{2}$.\\
A forma quadrática é a versão compacta e algébrica do balanço de potência, e
generaliza para qualquer número de malhas.}
\end{questao}

\begin{questao}[3]{}
As três malhas em linha formam uma escada: o condutor superior passa pelos nós
$A$, $B$, $C$, $D$; os ramos verticais em $B$ e $C$ são os compartilhados.
Com $I_1=\SI{3.75}{\ampere}$, $I_2=\SI{1.5}{\ampere}$ e
$I_3=\SI{0.75}{\ampere}$:
\begin{itensq}
\item Determine a corrente real em cada ramo.
\item Verifique a LKC nos nós $B$ e $C$.
\item Explique por que a verificação era desnecessária.
\end{itensq}
\resposta{$3{,}75$; $2{,}25$; $1{,}5$; $0{,}75$ e $\SI{0.75}{\ampere}$; a LKC é
satisfeita \emph{identicamente}}
\resolucao{(a) Ramos exclusivos conduzem a própria corrente de malha; ramos
compartilhados, a diferença:
\begin{center}
\begin{tabular}{ll}
\hline
Ramo & Corrente\\
\hline
horizontal $A$--$B$ (só malha 1) & $I_1=\SI{3.75}{\ampere}$\\
vertical em $B$ (malhas 1 e 2) & $I_1-I_2=\SI{2.25}{\ampere}$\\
horizontal $B$--$C$ (só malha 2) & $I_2=\SI{1.5}{\ampere}$\\
vertical em $C$ (malhas 2 e 3) & $I_2-I_3=\SI{0.75}{\ampere}$\\
horizontal $C$--$D$ (só malha 3) & $I_3=\SI{0.75}{\ampere}$\\
\hline
\end{tabular}
\end{center}
(b) \textbf{Nó $B$} --- chega o ramo $A$--$B$, saem o vertical e o
$B$--$C$:
\[
3{,}75=2{,}25+1{,}5.\ \checkmark
\]
\textbf{Nó $C$:} $1{,}5=0{,}75+0{,}75$ \checkmark.\\
(c) \textbf{Porque a LKC é uma identidade algébrica, não um resultado.}
Em $B$:
\[
I_1=(I_1-I_2)+I_2,
\]
verdadeiro para \emph{quaisquer} valores de $I_1$ e $I_2$ --- inclusive
errados. O mesmo em $C$: $I_2=(I_2-I_3)+I_3$.\\
\textbf{Razão estrutural:} cada corrente de malha percorre um \emph{laço
fechado}, entrando e saindo de todo nó que atravessa. Qualquer combinação de
correntes de malha satisfaz a lei dos nós por construção.\\
\textbf{Consequência conceitual:} as correntes de malha formam uma
\textbf{base} do espaço das correntes de ramo que obedecem à LKC. O método impõe
a LKC pela \emph{escolha das variáveis} e usa as equações apenas para a LKT ---
exatamente o inverso da análise nodal, que impõe a LKT ao adotar potenciais e
gasta as equações com a LKC.\\
\textbf{Alerta prático:} justamente por ser identidade, a LKC \emph{não serve}
como verificação em análise de malhas --- ela fecharia mesmo com o sistema
resolvido errado. Verifique sempre pela LKT, ou pelo balanço de potência.}
\end{questao}

\begin{questao}[3]{}
Um circuito de duas malhas produziu a matriz
$\begin{bmatrix}10&-10\\-10&10\end{bmatrix}$.
\begin{itensq}
\item Mostre que ela é singular.
\item Determine o que essa estrutura significa fisicamente.
\item Indique como corrigir a análise.
\end{itensq}
\resposta{$\det=0$; as duas malhas compartilham \emph{toda} a sua resistência}
\resolucao{(a) $\det=10(10)-(-10)(-10)=100-100=0$. O sistema não tem solução
única.\\
(b) $R_{11}=R_{22}=|R_{12}|=\SI{10}{\ohm}$ significa que a resistência
compartilhada é \emph{toda} a resistência de cada malha --- nenhuma delas possui
resistência própria. As duas "malhas" são, na verdade, o mesmo caminho de
corrente: elas não são independentes.\\
Formalmente, a matriz perdeu a dominância diagonal estrita, e as duas linhas são
proporcionais.\\
(c) \textbf{Causas prováveis e correções.}
\begin{itemize}
\item \textbf{Contornos mal escolhidos:} duas malhas que descrevem o mesmo laço.
Refaça a identificação, conferindo $m=b-n+1$.
\item \textbf{Resistência própria omitida:} um resistor exclusivo esquecido na
diagonal. Verifique a montagem elemento a elemento.
\item \textbf{Fonte de corrente ideal em ramo compartilhado} tratada como se
fosse resistência nula --- neste caso, a correção é usar supermalha.
\item \textbf{Circuito realmente desconexo} em duas partes sem acoplamento
resistivo, o que exigiria tratá-las separadamente.
\end{itemize}
\textbf{Diagnóstico rápido:} em rede passiva bem montada, a matriz de malhas é
\emph{sempre} simétrica e diagonal dominante. Determinante nulo denuncia erro de
montagem, não peculiaridade do circuito.}
\end{questao}

\begin{questao}[4]{}
\textbf{(Demonstração --- forma quadrática da potência.)} Prove que, na
formulação de malhas, a potência total dissipada vale
$P=\mathbf{I}^{\!\top}\mathbf{R}\,\mathbf{I}$.
\begin{itensq}
\item Demonstre para uma rede genérica de $m$ malhas.
\item Mostre que a expressão é sempre não negativa em rede passiva.
\item Relacione com o balanço $P=\mathbf{I}^{\!\top}\mathbf{b}$.
\end{itensq}
\resposta{$P=\sum_k R_{kk}I_k^{2}+2\sum_{j<k}R_{jk}I_jI_k$, forma quadrática
definida positiva}
\resolucao{(a) A potência é a soma sobre os \emph{ramos}. Um resistor
exclusivo da malha $k$ dissipa $R\,I_k^{2}$; um resistor $R_c$ compartilhado
pelas malhas $j$ e $k$ dissipa
\[
R_c(I_j-I_k)^{2}=R_cI_j^{2}+R_cI_k^{2}-2R_cI_jI_k.
\]
Ora, $R_c$ está incluído em $R_{jj}$ \emph{e} em $R_{kk}$, e
$R_{jk}=-R_c$. Somando sobre todos os ramos, os termos se reagrupam
exatamente como
\[
P=\sum_{k}R_{kk}I_k^{2}+2\sum_{j<k}R_{jk}I_jI_k
=\mathbf{I}^{\!\top}\mathbf{R}\,\mathbf{I}.
\]
(b) Cada parcela da soma original é da forma $R\,(\text{corrente})^{2}$ com
$R>0$, logo $P\ge0$ para \emph{qualquer} $\mathbf{I}$, anulando-se apenas com
$\mathbf{I}=\mathbf{0}$. Isso significa que $\mathbf{R}$ é \textbf{definida
positiva} --- e, em particular, não singular. \textbf{A física garante a
álgebra:} a existência e unicidade da solução decorre de a rede não poder
dissipar potência negativa.\\
(c) Multiplicando $\mathbf{R}\mathbf{I}=\mathbf{b}$ por
$\mathbf{I}^{\!\top}$ à esquerda:
\[
\mathbf{I}^{\!\top}\mathbf{R}\,\mathbf{I}=\mathbf{I}^{\!\top}\mathbf{b}
\;\Longrightarrow\;
P_{dissipada}=\sum_k b_kI_k=P_{fornecida}.
\]
O balanço de potência não é uma verificação \emph{adicional} --- ele é uma
consequência algébrica direta do sistema de malhas. É por isso que ele
\emph{sempre} fecha quando a montagem e a resolução estão corretas.}
\end{questao}

\begin{questao}[4]{}
\textbf{(Dualidade formal.)} Estabeleça a correspondência completa entre
análise nodal e análise de malhas.
\begin{itensq}
\item Construa a tabela de correspondências entre grandezas e estruturas.
\item Mostre que o mesmo circuito, transformado em seu dual, troca um método
      pelo outro.
\item Indique a limitação dessa dualidade.
\end{itensq}
\resposta{Tensão$\leftrightarrow$corrente,
$\mathbf{G}\leftrightarrow\mathbf{R}$, nó$\leftrightarrow$malha,
supernó$\leftrightarrow$supermalha; limitada a circuitos planares}
\resolucao{(a)
\begin{center}
\begin{tabular}{ll}
\hline
Análise nodal & Análise de malhas\\
\hline
Tensões nodais $\mathbf{v}$ & Correntes de malha $\mathbf{I}$\\
Matriz de condutâncias $\mathbf{G}$ & Matriz de resistências $\mathbf{R}$\\
$G_{kk}=\sum$ condutâncias do nó & $R_{kk}=\sum$ resistências da malha\\
$G_{jk}=-G_{comum}$ & $R_{jk}=-R_{comum}$\\
Fontes de corrente em $\mathbf{b}$ & Fontes de tensão em $\mathbf{b}$\\
Lei dos nós (LKC) & Lei das malhas (LKT)\\
Supernó (fonte de tensão flutuante) & Supermalha (fonte de corrente comum)\\
Nó de referência & Malha externa\\
$n-1$ equações & $b-n+1$ equações\\
\hline
\end{tabular}
\end{center}
(b) \textbf{Circuito dual:} substitua cada resistência $R$ por uma condutância
de mesmo valor numérico, cada fonte de tensão por uma de corrente (mesmo valor
numérico), série por paralelo, e associe a cada malha do original um nó do dual.
As equações nodais do dual são \emph{idênticas}, termo a termo, às equações de
malha do original.\\
\textbf{Consequência:} qualquer resultado provado para um método vale
automaticamente para o outro. A simetria e a definição positiva de
$\mathbf{R}$, por exemplo, são a mesma propriedade já estabelecida para
$\mathbf{G}$.\\
(c) \textbf{Limitação:} o dual só existe para circuitos \textbf{planares} ---
aqueles que podem ser desenhados sem cruzamentos. A análise de malhas
compartilha essa restrição, pois "malha" só se define em desenho plano.\\
A análise \textbf{nodal não tem essa limitação}: ela funciona em qualquer
topologia. É a razão prática pela qual simuladores como o SPICE são construídos
sobre formulação nodal, e não de malhas.}
\end{questao}

\begin{questao}[4]{}
\textbf{(Algoritmo.)} Formule um procedimento sistemático para gerar as
equações de malha a partir da descrição de um circuito planar.
\begin{itensq}
\item Descreva o algoritmo, incluindo o tratamento de fontes de corrente.
\item Determine o custo computacional.
\item Aponte a etapa mais difícil de automatizar e por quê.
\end{itensq}
\resposta{Identificação das faces do grafo planar é a etapa crítica; custo
$\mathcal{O}(m^{3})$ na resolução}
\resolucao{(a) \textbf{Algoritmo.}
\begin{enumerate}
\item Verificar planaridade e obter um \emph{mergulho} plano do grafo.
\item Identificar as $m=b-n+1$ faces internas (as malhas); a face externa é
      descartada.
\item Adotar sentido horário para todas as correntes de malha.
\item Inicializar $\mathbf{R}=\mathbf{0}$ e $\mathbf{b}=\mathbf{0}$.
\item Para cada resistor $R$: somar $R$ à diagonal de \emph{cada} malha que o
      contenha; se pertencer a duas malhas $j,k$, subtrair $R$ de
      $R_{jk}$ e de $R_{kj}$.
\item Para cada fonte de tensão: somar $\pm V$ em $b_k$ de cada malha que a
      contenha, conforme o sentido.
\item \textbf{Fontes de corrente:} se pertencer a uma só malha, fixar
      $I_k$ e eliminar a linha e a coluna correspondentes. Se for
      compartilhada, fundir as duas malhas em uma supermalha e acrescentar a
      equação de restrição $I_j-I_k=I_{fonte}$.
\item Resolver $\mathbf{R}\mathbf{I}=\mathbf{b}$ e expandir para as correntes
      de ramo.
\end{enumerate}
(b) \textbf{Custo:} montagem em $\mathcal{O}(b)$ --- cada elemento é visitado
uma vez. Resolução em $\mathcal{O}(m^{3}/3)$ por eliminação; muito menos se a
matriz for esparsa, o que é típico. O gargalo é a resolução, não a montagem.\\
(c) \textbf{A etapa crítica é a 2:} identificar as faces internas do grafo. Ela
exige um \emph{mergulho} plano --- uma escolha de como desenhar o grafo --- e não
é uma propriedade combinatória simples da lista de ramos. O teste de
planaridade é $\mathcal{O}(n)$, mas a implementação é notoriamente delicada.\\
Pior: se o circuito \textbf{não for planar}, o algoritmo simplesmente não tem
saída --- não há malhas a identificar.\\
\textbf{Conclusão:} a análise nodal dispensa toda a etapa 1--2, pois nós são
dados diretamente pela netlist. Essa é a razão de fundo pela qual a
formulação nodal venceu na prática computacional, apesar de a análise de malhas
ser frequentemente mais econômica no papel.}
\end{questao}

\begin{questao}[4]{}
\textbf{(Sensibilidade e matriz inversa.)} Para o circuito de duas malhas com
$\mathbf{R}=\begin{bmatrix}12&-6\\-6&12\end{bmatrix}$:
\begin{itensq}
\item Calcule $\mathbf{R}^{-1}$ e interprete cada elemento.
\item Determine como $I_1$ varia se a fonte da malha 2 aumentar em
      $\SI{6}{\volt}$.
\item Verifique a reciprocidade.
\end{itensq}
\resposta{$\mathbf{R}^{-1}=\frac{1}{108}\begin{bmatrix}12&6\\6&12\end{bmatrix}
\si{\per\ohm}$; $\Delta I_1=\SI{0.333}{\ampere}$}
\resolucao{(a) Com $\det=108$:
\[
\mathbf{R}^{-1}=\frac{1}{108}\begin{bmatrix}12&6\\6&12\end{bmatrix}
=\begin{bmatrix}0{,}1111&0{,}0556\\0{,}0556&0{,}1111\end{bmatrix}\si{\siemens}.
\]
O elemento $(\mathbf{R}^{-1})_{11}=\SI{0.111}{\siemens}$ é a
\emph{condutância de entrada} da malha 1: quantos ampères de $I_1$ por volt
aplicado nela. O elemento $(\mathbf{R}^{-1})_{12}=\SI{0.0556}{\siemens}$ é uma
\textbf{condutância de transferência}: quantos ampères de $I_1$ por volt
aplicado na malha \emph{2}.\\
(b)
\[
\Delta I_1=(\mathbf{R}^{-1})_{12}\,\Delta b_2=0{,}0556(6)=\SI{0.333}{\ampere}.
\]
\textbf{Verificação:} com $\mathbf{b}=(36;24)$,
$I_1=(36\cdot12+6\cdot24)/108=(432+144)/108=\SI{5.333}{\ampere}$ --- e o valor
anterior era $\SI{5}{\ampere}$ \checkmark.\\
(c) $(\mathbf{R}^{-1})_{12}=(\mathbf{R}^{-1})_{21}=\SI{0.0556}{\siemens}$:
aplicar $\SI{1}{\volt}$ na malha 2 e medir $I_1$ dá o mesmo que aplicar
$\SI{1}{\volt}$ na malha 1 e medir $I_2$.\\
\textbf{Origem:} a inversa de uma matriz simétrica é simétrica --- e
$\mathbf{R}$ só é simétrica porque não há fontes dependentes. Com o elemento
ativo da questão anterior, a reciprocidade se perderia, exatamente como ocorre
na formulação nodal.}
\end{questao}

\begin{questao}[4]{}
\textbf{(Escolha de método sob restrição.)} Uma rede planar tem $n$ nós e
$b$ ramos, com $f_i$ fontes de corrente e $f_v$ fontes de tensão.
\begin{itensq}
\item Estabeleça um critério quantitativo de escolha entre nodal e malhas.
\item Aplique a $n=6$, $b=10$, $f_i=4$, $f_v=1$.
\item Discuta o caso em que o critério e a contagem bruta discordam.
\end{itensq}
\resposta{Comparar $(n-1)$ e $(b-n+1)$ \emph{corrigidos} pelo efeito das
fontes; no exemplo, nodal com $1$ incógnita efetiva}
\resolucao{(a) \textbf{Critério corrigido.} O tamanho efetivo do sistema é o
número bruto de equações menos as incógnitas que as fontes \emph{eliminam}:
\begin{itemize}
\item \textbf{Nodal:} $(n-1)$ menos o número de fontes de tensão
\emph{aterradas} (cada uma fixa um potencial). Fontes de corrente não alteram o
tamanho --- entram direto no vetor $\mathbf{b}$.
\item \textbf{Malhas:} $(b-n+1)$ menos o número de fontes de corrente
\emph{na periferia} (cada uma fixa uma corrente de malha). Fontes de tensão
entram direto em $\mathbf{b}$.
\end{itemize}
Escolha o menor. Em caso de empate, prefira o método em que as fontes restantes
não exigem supernó ou supermalha.\\
(b) Nodal: $6-1=5$; com a fonte de tensão aterrada, $4$ incógnitas.\\
Malhas: $10-6+1=5$; se as quatro fontes de corrente estiverem na periferia,
restam $\mathbf{1}$ incógnita.\\
\textbf{Escolha: malhas}, com um sistema $1\times1$ --- resultado que a
contagem bruta (empate em $5$) não revelaria.\\
(c) \textbf{Quando discordam.} A contagem bruta é um limite \emph{superior};
o efeito das fontes pode reduzi-lo drasticamente e inverter a decisão, como
acabou de ocorrer.\\
\textbf{Mas há uma ressalva decisiva:} a redução em malhas depende de as fontes
de corrente poderem ser postas na periferia, o que nem sempre é possível --- com
quatro delas em uma rede de cinco malhas, é provável que ao menos uma fique
compartilhada, exigindo supermalha e devolvendo parte da vantagem.\\
\textbf{Procedimento honesto:} faça a contagem bruta, depois \emph{desenhe} os
contornos e verifique quantas fontes realmente caem na periferia. Só então
decida.}
\end{questao}

\subsubsection*{Fontes dependentes --- análise de malhas}

\blocoaprofundamento

\begin{questao}[3]{}
As correntes de malha $I_1$ e $I_2$ são horárias. A fonte dependente da malha 2 vale $2{,}5I_1$ volts e auxilia $I_2$. Determine $I_1$ e $I_2$.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{20}{\volt}$}] (0,3) to[R,l={$\SI{5}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[R,l={$\SI{5}{\ohm}$}] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{5}{\ohm}$}] (6,3) to[cvsource,l={$2{,}5I_1$}] (6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\SI{3.2}{\ampere}$; $I_2=\SI{2.4}{\ampere}$}
\resolucao{Malha 1: $10I_1-5I_2=20$. Malha 2: $-5I_1+10I_2=2{,}5I_1$, isto é, $-7{,}5I_1+10I_2=0$. A solução é $(3{,}2;2{,}4)$ A.}
\end{questao}

\begin{questao}[3]{}
A fonte controlada da malha 2 vale $3I_1$ volts e auxilia $I_2$. Determine as correntes de malha.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{24}{\volt}$}] (0,3) to[R,l={$\SI{4}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[R,l={$\SI{2}{\ohm}$}] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{6}{\ohm}$}] (6,3) to[cvsource,l={$3I_1$}] (6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\dfrac{96}{19}\,\si{\ampere}\approx\SI{5.05}{\ampere}$; $I_2=\dfrac{60}{19}\,\si{\ampere}\approx\SI{3.16}{\ampere}$}
\resolucao{$6I_1-2I_2=24$ e $-2I_1+8I_2=3I_1$. Portanto $-5I_1+8I_2=0$ e a solução é $I_1=96/19$ A, $I_2=60/19$ A.}
\end{questao}

\begin{questao}[3]{}
Na periferia da malha 2 há uma fonte de corrente dependente que impõe $I_2=0{,}5i_x$, com $i_x=I_1-I_2$ no resistor compartilhado. Determine $I_1$ e $I_2$.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{12}{\volt}$}] (0,3) to[R,l={$\SI{2}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[R,l={$\SI{4}{\ohm}$},i>^={$i_x$}] (3,0) coordinate(b)--(0,0);
\draw (t) -- (6,3) to[cisource,l={$0{,}5i_x$}] (6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\dfrac{18}{7}\,\si{\ampere}\approx\SI{2.57}{\ampere}$; $I_2=\dfrac{6}{7}\,\si{\ampere}\approx\SI{0.857}{\ampere}$}
\resolucao{$I_2=0{,}5(I_1-I_2)\Rightarrow I_1=3I_2$. Na malha 1, $2I_1+4(I_1-I_2)=12$. Substituindo $I_1=3I_2$, obtém-se $14I_2=12$.}
\end{questao}

\begin{questao}[3]{}
A fonte de corrente dependente está no ramo comum e aponta para baixo, impondo $I_1-I_2=0{,}5I_1$. Use supermalha para determinar $I_1$ e $I_2$.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{30}{\volt}$}] (0,3) to[R,l={$\SI{6}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[cisource,l={$0{,}5I_1$},invert] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{6}{\ohm}$}] (6,3)--(6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\SI{3.33}{\ampere}$; $I_2=\SI{1.67}{\ampere}$}
\resolucao{Restrição: $I_1-I_2=0{,}5I_1\Rightarrow I_2=0{,}5I_1$. Supermalha externa: $6I_1+6I_2=30$. Logo $9I_1=30$.}
\end{questao}

\blocodesafio

\begin{questao}[4]{}
A fonte de tensão controlada está no ramo comum e tem polaridade positiva no topo, com valor $2I_2$ volts. As fontes independentes impulsionam as correntes horárias. Determine $I_1$ e $I_2$.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{20}{\volt}$}] (0,3) to[R,l={$\SI{4}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[cvsource,l={$2I_2$}] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{4}{\ohm}$}] (6,3) to[battery1,l={$\SI{8}{\volt}$},invert] (6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\SI{3}{\ampere}$; $I_2=\SI{4}{\ampere}$}
\resolucao{Na malha 1, $4I_1+2I_2=20$. Na malha 2, a fonte controlada é atravessada no sentido oposto: $4I_2-2I_2=8$, logo $I_2=4$ A; então $I_1=3$ A.}
\end{questao}

\begin{questao}[4]{}
A fonte de corrente no ramo comum vale $2i_x$, onde $i_x=I_2$ é a corrente no resistor externo da malha 2. Determine as correntes usando supermalha.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{36}{\volt}$}] (0,3) to[R,l={$\SI{4}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[cisource,l={$2i_x$},invert] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{8}{\ohm}$},i>^={$i_x$}] (6,3)--(6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$I_1=\SI{5.4}{\ampere}$; $I_2=\SI{1.8}{\ampere}$}
\resolucao{No ramo comum, $I_1-I_2=2I_2\Rightarrow I_1=3I_2$. Na supermalha: $4I_1+8I_2=36$. Assim $20I_2=36$.}
\end{questao}

\begin{questao}[4]{}
Três malhas em linha têm resistores próprios de $\SI{4}{\ohm}$ e resistores compartilhados de $\SI{2}{\ohm}$. A malha 1 possui uma fonte de $\SI{24}{\volt}$; a malha 3 possui uma fonte dependente de $2I_1$ volts que auxilia $I_3$. Determine $I_1$, $I_2$ e $I_3$.
\circuitoq{\begin{circuitikz}[american,scale=.72,transform shape]
\draw (0,0) to[battery1,l_={$\SI{24}{\volt}$}] (0,3) to[R,l={$\SI{4}{\ohm}$}] (2.6,3) coordinate(a);
\draw (a) to[R,l={$\SI{2}{\ohm}$}] (2.6,0) coordinate(b)--(0,0);
\draw (a) to[R,l={$\SI{4}{\ohm}$}] (5.2,3) coordinate(c);
\draw (c) to[R,l={$\SI{2}{\ohm}$}] (5.2,0) coordinate(d)--(b);
\draw (c) to[R,l={$\SI{4}{\ohm}$}] (7.8,3) to[cvsource,l={$2I_1$}] (7.8,0)--(d);
\node[loopcol] at(1.3,1.5){$I_1$};\node[loopcol] at(3.9,1.5){$I_2$};\node[loopcol] at(6.5,1.5){$I_3$};
\end{circuitikz}}
\resposta{$I_1=\dfrac{132}{29}\,\si{\ampere}\approx\SI{4.55}{\ampere}$; $I_2=\dfrac{48}{29}\,\si{\ampere}\approx\SI{1.66}{\ampere}$; $I_3=\dfrac{60}{29}\,\si{\ampere}\approx\SI{2.07}{\ampere}$}
\resolucao{O sistema é $6I_1-2I_2=24$, $-2I_1+8I_2-2I_3=0$ e $-2I_2+6I_3=2I_1$. Resolvendo, obtêm-se $I_1=132/29$, $I_2=48/29$ e $I_3=60/29$ A.}
\end{questao}

\begin{questao}[4]{}
No circuito simétrico, a fonte dependente da malha 2 vale $kI_1$ volts e auxilia $I_2$. Determine $k$ para que $I_2=\SI{3}{\ampere}$.
\circuitoq{\begin{circuitikz}[american,scale=.9,transform shape]
\draw (0,0) to[battery1,l_={$\SI{30}{\volt}$}] (0,3) to[R,l={$\SI{5}{\ohm}$}] (3,3) coordinate(t);
\draw (t) to[R,l={$\SI{5}{\ohm}$}] (3,0) coordinate(b)--(0,0);
\draw (t) to[R,l={$\SI{5}{\ohm}$}] (6,3) to[cvsource,l={$kI_1$}] (6,0)--(b);
\draw[->,loopcol] (1.5,1.5) arc (0:300:.55);\node[loopcol] at(1.5,1.5){$I_1$};
\draw[->,loopcol] (4.7,1.5) arc (0:300:.55);\node[loopcol] at(4.7,1.5){$I_2$};
\end{circuitikz}}
\resposta{$k=\dfrac{5}{3}\,\ohm\approx\SI{1.67}{\ohm}$}
\resolucao{Com $I_2=3$ A, a malha 1 dá $10I_1-5(3)=30\Rightarrow I_1=4{,}5$ A. A malha 2 dá $-5I_1+10I_2=kI_1$, logo $-22{,}5+30=4{,}5k$ e $k=5/3\,\ohm$.}
\end{questao}
